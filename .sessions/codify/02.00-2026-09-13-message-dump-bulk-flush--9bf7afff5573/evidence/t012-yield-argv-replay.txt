"""Independent Yield replay for T012 (finding seam-S6-value-word-split, C2).

Stands a PATH shim in for the `buster-ripper` console script, runs the wrapper
hermetically (no server, no port bound, no upstream contacted), and prints the
raw argv the wrapper assembled. Every case runs against BOTH the post-task bytes
and the pre-task bytes so (a) the new witnesses are shown to be fail-first rather
than decorative, and (b) the documented forms' word sets can be compared for
byte-identity across the increment.
"""

import os
import subprocess
import sys
import tempfile
from pathlib import Path

REPO = Path("/Users/alexhubitski/Projects/buster-ripper")
NEW_BYTES = (REPO / "serve_buster_ripper.sh").read_bytes()
OLD_BYTES = subprocess.run(
    ["git", "show", "0c38a8e:serve_buster_ripper.sh"],
    cwd=REPO, capture_output=True, check=True,
).stdout

CONSUMED = (
    "UPSTREAM", "HOST", "PORT", "STRIP_DATE", "VERBOSE", "SKIP_TLS_VERIFY",
    "DUMP_DIR", "DUMP_FORMAT", "PRESERVED_THINKING",
)
ARGV_OUT = "BUSTER_RIPPER_ARGV_OUT"
SHIM = "#!/bin/sh\nprintf '%s\\0' \"$@\" > \"$BUSTER_RIPPER_ARGV_OUT\"\nexit 0\n"


def run_case(script: Path, workdir: Path, **overrides):
    """Run one wrapper invocation in `workdir` (its cwd) and record argv."""
    (workdir / "argv").unlink(missing_ok=True)
    env = {k: v for k, v in os.environ.items() if k not in CONSUMED}
    env["PATH"] = f"{workdir / 'bin'}{os.pathsep}{env['PATH']}"
    env["HOME"] = str(workdir / "home")
    env[ARGV_OUT] = str(workdir / "argv")
    for key, val in overrides.items():
        if val is None:
            env.pop(key, None)
        else:
            env[key] = val
    (workdir / "bin").mkdir(exist_ok=True)
    shim = workdir / "bin" / "buster-ripper"
    shim.write_text(SHIM)
    shim.chmod(0o755)
    r = subprocess.run(["bash", str(script)], env=env, cwd=str(workdir),
                       capture_output=True, text=True, check=False)
    argv = []
    out = workdir / "argv"
    if out.exists():
        fields = out.read_bytes().split(b"\0")
        if fields and fields[-1] == b"":
            fields.pop()
        argv = [f.decode("utf-8") for f in fields]
    return r.returncode, r.stderr.strip(), argv


CASES = [
    ("space-bearing path  DUMP_DIR=/tmp/my docs", dict(DUMP_DIR="/tmp/my docs")),
    ("whitespace-only     DUMP_DIR=' '", dict(DUMP_DIR=" ")),
    ("whitespace-only     DUMP_DIR=$'\\t'", dict(DUMP_DIR="\t")),
    ("format with space   DUMP_FORMAT='a b'", dict(DUMP_DIR="", DUMP_FORMAT="a b")),
    ("both values spaced", dict(DUMP_DIR="/tmp/my docs", DUMP_FORMAT="change focused")),
    ("glob pattern        DUMP_DIR='*.md'", dict(DUMP_DIR="*.md")),
    ("flag-shaped value   DUMP_DIR='--verbose'", dict(DUMP_DIR="--verbose")),
    ("ALL SUPPRESSED", dict(DUMP_DIR="", DUMP_FORMAT="", PRESERVED_THINKING="0")),
    ("T011 state: unset", dict(DUMP_DIR=None)),
    ("T011 state: empty", dict(DUMP_DIR="")),
    ("T011 state: set", dict(DUMP_DIR="/tmp/served-diffs")),
    ("README  STRIP_DATE=1", dict(DUMP_DIR="", STRIP_DATE="1")),
    ("README  VERBOSE=1", dict(DUMP_DIR="", VERBOSE="1")),
    ("README  SKIP_TLS_VERIFY=1", dict(DUMP_DIR="", SKIP_TLS_VERIFY="1")),
    ("README  DUMP_FORMAT=change-focused", dict(DUMP_DIR="", DUMP_FORMAT="change-focused")),
    ("README  UPSTREAM/PORT overrides", dict(DUMP_DIR="", UPSTREAM="http://10.0.0.1:30000", PORT="30002")),
    ("README  header DUMP_DIR=\"\" disable", dict(DUMP_DIR="")),
]

DOCUMENTED = {"README  STRIP_DATE=1", "README  VERBOSE=1", "README  SKIP_TLS_VERIFY=1",
              "README  DUMP_FORMAT=change-focused", "README  UPSTREAM/PORT overrides",
              "README  header DUMP_DIR=\"\" disable", "T011 state: unset",
              "T011 state: empty", "T011 state: set"}


def sweep(label, script, root):
    """Run every case once; return {name: (rc, stderr, argv)} and print reprs."""
    print(f"\n########## {label} ##########")
    results = {}
    for name, over in CASES:
        case_dir = root / name.split()[0].strip() / str(abs(hash(name)))
        case_dir.mkdir(parents=True, exist_ok=True)
        (case_dir / "a.md").write_text("")
        (case_dir / "b.md").write_text("")
        rc, err, argv = run_case(script, case_dir, **over)
        argv = [w.replace(str(case_dir / "home"), "$HOME") for w in argv]
        results[name] = (rc, err, argv)
        print(f"{name:44s} rc={rc} argc={len(argv)}")
        print(f"{'':44s}   {argv!r}")
        if err:
            print(f"{'':44s}   stderr: {err}")
    return results


def main() -> int:
    import hashlib
    for exe in (["bash"], ["env", "bash"], ["/bin/bash"]):
        out = subprocess.run(exe + ["--version"], capture_output=True, text=True).stdout
        print(f"{' '.join(exe):12s} -> {out.splitlines()[0]}")
    print(f"new script sha256 {hashlib.sha256(NEW_BYTES).hexdigest()[:16]}")
    print(f"old script sha256 {hashlib.sha256(OLD_BYTES).hexdigest()[:16]}")

    with tempfile.TemporaryDirectory() as d:
        root = Path(d)
        new_script = root / "new.sh"
        new_script.write_bytes(NEW_BYTES)
        old_script = root / "old.sh"
        old_script.write_bytes(OLD_BYTES)
        new_res = sweep(f"POST-TASK bytes ({hashlib.sha256(NEW_BYTES).hexdigest()[:8]})", new_script, root)
        old_res = sweep(f"PRE-TASK bytes  ({hashlib.sha256(OLD_BYTES).hexdigest()[:8]})", old_script, root)

    print("\n########## COMPARISON ##########")
    changed = [n for n, _, in CASES if new_res[n][2] != old_res[n][2] or new_res[n][0] != old_res[n][0]]
    print("cases whose observable moved (fail-first candidates):")
    for n in changed:
        print(f"  - {n}")
        print(f"      old: rc={old_res[n][0]} {old_res[n][2]!r}")
        print(f"      new: rc={new_res[n][0]} {new_res[n][2]!r}")
    print("\ndocumented / T011 forms (must be IDENTICAL across the increment):")
    for n in sorted(DOCUMENTED):
        same = new_res[n][2] == old_res[n][2] and new_res[n][0] == old_res[n][0]
        print(f"  {'OK  ' if same else 'DIFF'}  {n}  argc={len(new_res[n][2])}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
